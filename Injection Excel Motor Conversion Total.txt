# ============================================================
# CONFIGURAÇÃO
# ============================================================

import re
import pandas as pd

from pyspark.sql.functions import (
    current_timestamp,
    lit
)

catalog_name = "agpt_insurance_dv_catalog"
schema_name = "wkb_bus_operations_claimsmotor"
table_name = "raw_bda_agentes_sas"

full_table_name = f"{catalog_name}.{schema_name}.{table_name}"

source_file = "/Volumes/agpt_insurance_dv_catalog/wkb_bus_operations_claimsmotor/pba&uw/01_raw/BDA_AGENTES.xlsx"

print(f"Tabela destino: {full_table_name}")
print(f"Ficheiro origem: {source_file}")

# ============================================================
# FUNÇÕES AUXILIARES
# ============================================================

def decode_excel_column_name(name):

    if name is None:
        return ""

    return re.sub(
        r"_x([0-9A-Fa-f]{4})_",
        lambda m: chr(int(m.group(1), 16)),
        str(name)
    )


def sanitize_column_name(name):

    novo_nome = decode_excel_column_name(name)

    substituicoes = {
        "º": "o",
        "ª": "a",
        "%": "pct",
        "/": "_",
        "\\": "_",
        "#": "num",
        "€": "eur",
        "$": "usd",
        "&": "and",
        "-": "_",
        ".": "_",
        ":": "_"
    }

    for original, substituto in substituicoes.items():
        novo_nome = novo_nome.replace(original, substituto)

    novo_nome = re.sub(r"[ ,;{}()\n\t=]", "_", novo_nome)
    novo_nome = re.sub(r"_+", "_", novo_nome)

    novo_nome = novo_nome.strip("_").lower()

    if not novo_nome:
        novo_nome = "coluna_sem_nome"

    return novo_nome


def sanitize_dataframe_columns(df):

    nomes_originais = df.columns
    nomes_novos = []
    contador = {}

    for nome in nomes_originais:

        nome_limpo = sanitize_column_name(nome)

        if nome_limpo in contador:
            contador[nome_limpo] += 1
            nome_limpo = f"{nome_limpo}_{contador[nome_limpo]}"
        else:
            contador[nome_limpo] = 0

        nomes_novos.append(nome_limpo)

    for original, novo in zip(nomes_originais, nomes_novos):

        if original != novo:
            df = df.withColumnRenamed(original, novo)

    return df


# ============================================================
# VALIDAR CATÁLOGO
# ============================================================

catalog_exists = (
    spark.sql("SHOW CATALOGS")
         .filter(f"catalog = '{catalog_name}'")
         .count()
)

if catalog_exists == 0:
    raise Exception(f"Catálogo não encontrado: {catalog_name}")

print(f"OK - Catálogo encontrado: {catalog_name}")

# ============================================================
# VALIDAR SCHEMA
# ============================================================

schema_exists = (
    spark.sql(f"SHOW SCHEMAS IN {catalog_name}")
         .filter(f"databaseName = '{schema_name}'")
         .count()
)

if schema_exists == 0:
    raise Exception(
        f"Schema não encontrado: {catalog_name}.{schema_name}"
    )

print(f"OK - Schema encontrado: {schema_name}")

# ============================================================
# LEITURA EXCEL
# ============================================================

print("A ler Excel...")

pdf = pd.read_excel(
    source_file,
    engine="openpyxl"
)

pdf = pdf.where(pd.notnull(pdf), None)

print(f"Registos encontrados: {len(pdf):,}")

if len(pdf) == 0:
    raise Exception("Ficheiro sem dados.")

# ============================================================
# CRIAR DATAFRAME SPARK
# ============================================================

df = spark.createDataFrame(pdf)

# ============================================================
# SANITIZAR COLUNAS
# ============================================================

df = sanitize_dataframe_columns(df)

print("Schema inferido:")

df.printSchema()

# ============================================================
# COLUNAS TÉCNICAS
# ============================================================

df = (
    df
    .withColumn(
        "source_file",
        lit("BDA_AGENTES.xlsx")
    )
    .withColumn(
        "load_timestamp",
        current_timestamp()
    )
)

# ============================================================
# DROP TABELA EXISTENTE
# ============================================================

spark.sql(
    f"DROP TABLE IF EXISTS {full_table_name}"
)

print("Tabela anterior removida.")

# ============================================================
# ESCRITA DELTA
# ============================================================

(
    df.write
      .format("delta")
      .mode("overwrite")
      .option("overwriteSchema", "true")
      .saveAsTable(full_table_name)
)

print("Tabela criada.")

# ============================================================
# VALIDAÇÃO
# ============================================================

df_final = spark.table(full_table_name)

print(f"Registos carregados: {df_final.count():,}")

df_final.printSchema()

display(df_final.limit(100))

print("Carga concluída com sucesso.")
