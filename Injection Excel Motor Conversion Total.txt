def log(msg=""):

def timed(label):
    class _T:
        def __enter__(self):
            self.t0 = time.time()
            return self

        def __exit__(self, *args):
            log("  [tempo] " + label + ": " + "{:.1f}s".format(time.time() - self.t0))

    return _T()

_XML_ESCAPE_RE = re.compile(r"_x([0-9A-Fa-f]{4,8})_")
_XML_ESCAPE_LOOSE_RE = re.compile(r"(?<![A-Za-z0-9])x([0-9A-Fa-f]{4})(?![A-Za-z0-9])")


def decode_excel_xml_escapes(value):
    if value is None or not isinstance(value, str):
        return value
    text = value

    def _to_char(match):
        try:
            return chr(int(match.group(1), 16))
        except (ValueError, OverflowError):
            return match.group(0)

    text = _XML_ESCAPE_RE.sub(_to_char, text)
    if "x00" in text:
        text = _XML_ESCAPE_LOOSE_RE.sub(_to_char, text)
    text = text.replace("\xa0", " ")
    text = text.replace("\u200b", "")
    return text


def decode_cell(value):
    if isinstance(value, str):
        text = decode_excel_xml_escapes(value).strip()
        if text == "":
            return None
        return text
    return value


def _needs_xml_decode(text):
    return ("_x" in text) or ("x00" in text) or ("\xa0" in text) or ("\u200b" in text)


def decode_cell_fast(value):
    if not isinstance(value, str):
        return value
    if not _needs_xml_decode(value):
        text = value.strip()
        return text if text else None
    return decode_cell(value)
