_XML_ESCAPE_RE = re.compile(r"_x([0-9A-Fa-f]{4,8})_", re.IGNORECASE)


def decode_excel_xml(value):
    if value is None or not isinstance(value, str):
        return value

    def _to_char(match):
        try:
            return chr(int(match.group(1), 16))
        except (ValueError, OverflowError):
            return match.group(0)

    text = _XML_ESCAPE_RE.sub(_to_char, value)
    return text.replace("\xa0", " ").replace("\u200b", "")


def sanitize_column_name(name):
    nome = decode_excel_xml(str(name) if name is not None else "")
    nome = nome.strip()
    if nome.lower().startswith("unnamed") or nome.lower() in ("", "nan", "none", "nat"):
        return "coluna_sem_nome"
    for old, new in {
        "º": "o", "ª": "a", "%": "pct", "/": "_", "\\": "_", "#": "num",
        "€": "eur", "$": "usd", "&": "e", "-": "_", ".": "_", ":": "_",
        "?": "", "!": "", "*": "", "+": "mais", "'": "", '"': "",
    }.items():
        nome = nome.replace(old, new)
    nome = re.sub(r"[ ,;{}()\n\t\r=]", "_", nome)
    nome = re.sub(r"_+", "_", nome).strip("_")
    nome = nome.lower()
    if nome == "":
        nome = "coluna_sem_nome"
    if nome[0].isdigit():
        nome = "c_" + nome
    return nome


def uniquify_names(names):
    usados, out = {}, []
    for nome in names:
        if nome not in usados:
            usados[nome] = 0
            out.append(nome)
        else:
            usados[nome] += 1
            out.append(nome + "_" + str(usados[nome]))
    return out


def _fold(name):
    n = unicodedata.normalize("NFD", str(name or ""))
    n = "".join(ch for ch in n if unicodedata.category(ch) != "Mn")
    return n.lower()


# ------------------------------------------------------------
# Classificacao de tipos pelo nome + amostra
# ------------------------------------------------------------

DATE_TOKENS = (
    "data", "dt_", "_dt", "nascim", "fnacim", "aniv",
    "inicio", "fim", "validade", "emissao", "registo",
)
INT_TOKENS = (
    "idade", "ano", "n_agente", "num_agente", "codigo_agente",
    "cod_agente", "n_agentes", "porta", "cp4", "codpostal_4",
)
ID_STRING_TOKENS = (
    "nif", "nipc", "niss", "iban", "email", "telefone", "telemovel",
    "cod_postal", "codigo_postal", "cdpostal", "cp7",
)


def classificar_coluna(nome, series=None):
    n = _fold(nome)
    if any(tok in n for tok in ID_STRING_TOKENS):
        return "string"
    if n == "fnacim" or any(tok in n for tok in DATE_TOKENS):
        return "date"
    if any(tok in n for tok in INT_TOKENS) and "data" not in n:
        return "int"
    if series is not None:
        amostra = [v for v in series.dropna().astype(str).head(80).tolist() if str(v).strip() not in ("", "nan", "None", "NaT")]
        if amostra:
            dateish = 0
            intish = 0
            for v in amostra:
                s = str(v).strip()
                if re.match(r"^\d{4}-\d{2}-\d{2}", s) or re.match(r"^\d{1,2}[/-]\d{1,2}[/-]\d{2,4}$", s):
                    dateish += 1
                elif re.match(r"^-?\d+(\.0+)?$", s):
                    intish += 1
            if dateish >= max(3, int(0.6 * len(amostra))):
                return "date"
            if intish >= max(3, int(0.8 * len(amostra))):
                return "int"
    return "string"


def _sql_ident(colname):
    return "`" + str(colname).replace("`", "") + "`"


def _sql_num(ident):
    raw = "regexp_replace(trim(cast(" + ident + " as string)), '[\\u00A0 €$]', '')"
    raw = "nullif(" + raw + ", '')"
    raw = (
        "case"
        " when " + raw + " is null then null"
        " when lower(" + raw + ") in ('nan','none','nat','null','-') then null"
        " when " + raw + " like '%,%' then regexp_replace(" + raw + ", ',', '.')"
        " else " + raw + " end"
    )
    raw = "regexp_replace(" + raw + ", '[^0-9.-]', '')"
    raw = "regexp_replace(" + raw + ", '\\\\.0+$', '')"
    raw = "regexp_replace(" + raw + ", '\\\\.$', '')"
    return "case when " + raw + " in ('', '.', '-', '-.') then null else " + raw + " end"


def aplicar_tipos_spark(df, kinds):
    selects = []
    for c in df.columns:
        ident = _sql_ident(c)
        kind = kinds.get(c, "string")
        if c in ("source_file", "source_sheet", "load_timestamp"):
            selects.append(ident)
            continue
        if kind == "date":
            txt = "trim(cast(" + ident + " as string))"
            selects.append(
                "coalesce("
                "try_cast(" + ident + " as date),"
                "to_date(" + txt + ", 'yyyy-MM-dd'),"
                "to_date(" + txt + ", 'dd/MM/yyyy'),"
                "to_date(" + txt + ", 'dd-MM-yyyy'),"
                "to_date(" + txt + ", 'yyyy/MM/dd'),"
                "to_date(" + txt + ", 'dd.MM.yyyy')"
                ") as " + ident
            )
        elif kind == "int":
            selects.append("try_cast(" + _sql_num(ident) + " as int) as " + ident)
        else:
            selects.append("cast(" + ident + " as string) as " + ident)
    return df.selectExpr(*selects)
