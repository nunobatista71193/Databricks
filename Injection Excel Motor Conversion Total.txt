def timed(label):
    class _T:
        def __enter__(self):
            self.t0 = time.time()
            return self
        def __exit__(self, *args):
            log("  [tempo] " + label + ": " + "{:.1f}s".format(time.time() - self.t0))
    return _T()

_XML_ESCAPE_RE = re.compile(r"_x([0-9A-Fa-f]{4,8})_")
_XML_ESCAPE_LOOSE_RE = re.compile(r"(?<![A-Za-z0-9])x([0-9A-Fa-f]{4})(?![A-Za-z0-9])")


def decode_excel_xml_escapes(value):
    if value is None or not isinstance(value, str):
        return value
    text = value
    def _to_char(match):
        try:
            return chr(int(match.group(1), 16))
        except (ValueError, OverflowError):
            return match.group(0)
    text = _XML_ESCAPE_RE.sub(_to_char, text)
    if "x00" in text:
        text = _XML_ESCAPE_LOOSE_RE.sub(_to_char, text)
    return text.replace("\xa0", " ").replace("\u200b", "")


def decode_cell(value):
    if isinstance(value, str):
        text = decode_excel_xml_escapes(value).strip()
        return text if text else None
    return value


def _needs_xml_decode(text):
    return ("_x" in text) or ("x00" in text) or ("\xa0" in text) or ("\u200b" in text)


def decode_object_column(series):
    if series.dtype != object:
        return series
    sample = series.dropna().head(50).tolist()
    has_str = has_escape = False
    for v in sample:
        if isinstance(v, str):
            has_str = True
            if _needs_xml_decode(v):
                has_escape = True
                break
    if not has_str:
        return series
    if has_escape:
        return series.map(decode_cell)
    return series


def sanitize_column_name(name):
    nome = decode_excel_xml_escapes(str(name) if name is not None else "").strip()
    if nome.lower().startswith("unnamed") or nome.lower() in ("", "nan", "none", "nat"):
        return "coluna_sem_nome"
    nome = re.sub(r"[ ,;{}()\n\t\r=]", "_", nome)
    for old, new in {
        "º": "o", "ª": "a", "%": "pct", "/": "_", "\\": "_", "#": "num",
        "€": "eur", "$": "usd", "-": "_", ".": "_", "?": "", "!": "",
        "*": "", "+": "mais", "&": "e", "'": "", '"': "",
    }.items():
        nome = nome.replace(old, new)
    nome = re.sub("_+", "_", nome).strip("_") or "coluna_sem_nome"
    if nome[0].isdigit():
        nome = "c_" + nome
    return nome


def uniquify_names(names):
    usados, resultado = {}, []
    for nome in names:
        if nome not in usados:
            usados[nome] = 0
            resultado.append(nome)
        else:
            usados[nome] += 1
            resultado.append(nome + "_" + str(usados[nome]))
    return resultado

def rename_ags_cdpostal(columns):
    novas, ja = [], False
    for nome in columns:
        if nome.upper() == "CDPOSTAL" and not ja:
            novas.append("CODIGO_POSTAL")
            ja = True
        else:
            novas.append(nome)
    return uniquify_names(novas)


def normalizar_codigo_postal_series(series):
    text = series.map(
        lambda v: None if v is None or (isinstance(v, float) and pd.isna(v))
        else str(int(v)) if isinstance(v, int) else str(v)
    ).astype("string").str.strip()
    text = text.str.replace("\xa0", "", regex=False).str.replace(" ", "", regex=False)
    text = text.str.replace(r"\.0+$", "", regex=True)
    text = text.mask(text.isna() | text.isin(["", "nan", "none", "nat", "NaN", "None", "NaT"]), pd.NA)
    out = pd.Series(pd.NA, index=series.index, dtype="string")
    out = out.mask(text.str.match(r"^\d{4}-\d{3}$", na=False), text)
    out = out.mask(text.str.match(r"^\d{7}$", na=False), text.str.slice(0, 4) + "-" + text.str.slice(4, 7))
    return out.astype(object).where(out.notna(), None)


def aplicar_regras_negocio(pdf, source_file):
    if is_ags_file(source_file):
        pdf.columns = rename_ags_cdpostal(list(pdf.columns))
        log("Regra AGS aplicada: CDPOSTAL -> CODIGO_POSTAL")
    if is_ocs_file(source_file):
        alvo = next((c for c in pdf.columns if c.upper() == "CODIGO_POSTAL"), None)
        if alvo is None:
            log("AVISO OCS: coluna CODIGO_POSTAL nao encontrada.")
        else:
            if alvo != "CODIGO_POSTAL":
                pdf = pdf.rename(columns={alvo: "CODIGO_POSTAL"})
                alvo = "CODIGO_POSTAL"
            pdf[alvo] = normalizar_codigo_postal_series(pdf[alvo])
            log(
                "Regra OCS aplicada em CODIGO_POSTAL: "
                + "{:,}".format(int(pdf[alvo].notna().sum()))
                + " validos / "
                + "{:,}".format(len(pdf))
                + " linhas"
            )
    return pdf

def classificar_coluna(nome):
    if nome in FORCE_STRING_COLS:
        return "string"
    if nome in FORCE_INT_COLS:
        return "int"
    if nome in FORCE_DECIMAL_COLS:
        return "decimal"
    if nome in META_COLS and nome != "header_row":
        return "meta"
    n = _fold(nome)
    if n in EXACT_TYPES:
        return EXACT_TYPES[n]
    if n in STRING_EXACT:
        return "string"
    if n.startswith(FLAG_PREFIXES) or n.endswith("_flag") or n.endswith("_flg"):
        return "flag"
    if any(tok in n for tok in STRING_TOKENS):
        return "string"
    if any(tok in n for tok in DECIMAL_TOKENS):
        return "decimal"
    if n == "idade" or n.startswith("idade_") or any(tok in n for tok in INT_TOKENS):
        return "int"
    return "string"


def _to_numeric_string(colname):
    s = F.trim(F.col(colname).cast("string"))
    s = F.regexp_replace(s, "[\u00a0 ]", "")
    s = F.regexp_replace(s, r"[€$]|eur|EUR", "")
    s = F.when(s.isin("", "nan", "None", "NONE", "NaT", "null", "NULL", "-"), F.lit(None)).otherwise(s)
    both = s.contains(".") & s.contains(",")
    only_comma = s.contains(",") & (~s.contains("."))
    s = (
        F.when(both, F.regexp_replace(F.regexp_replace(s, r"\.", ""), ",", "."))
        .when(only_comma, F.regexp_replace(s, ",", "."))
        .otherwise(s)
    )
    s = F.regexp_replace(s, r"[^0-9.\-]", "")
    return F.when(s.isin("", ".", "-", "-."), F.lit(None)).otherwise(s)


def _sql_ident(colname):
    return "`" + str(colname).replace("`", "") + "`"


def _sql_try_cast(colname, type_sql):
    return F.expr("try_cast(" + _sql_ident(colname) + " as " + type_sql + ")")


def aplicar_tipos_spark(df):
    decimais, inteiros, flags, textos = [], [], [], []
    for c in df.columns:
        kind = classificar_coluna(c)
        if kind == "meta":
            continue
        if kind == "decimal":
            decimais.append(c)
            df = df.withColumn(c, _to_numeric_string(c))
            df = df.withColumn(c, _sql_try_cast(c, "decimal(18,2)"))
        elif kind == "flag":
            flags.append(c)
            raw = F.upper(F.trim(F.col(c).cast("string")))
            raw = F.regexp_replace(raw, "[\u00a0 ]", "")
            mapped = (
                F.when(raw.isin("1", "1.0", "S", "SIM", "Y", "YES", "TRUE", "T", "X", "VERDADEIRO"), F.lit(1))
                .when(raw.isin("0", "0.0", "N", "NAO", "NÃO", "NO", "FALSE", "F", "FALSO", ""), F.lit(0))
                .otherwise(F.lit(None))
            )
            numeric = F.regexp_replace(_to_numeric_string(c), r"\.0+$", "")
            df = df.withColumn(c, F.coalesce(mapped, numeric))
            df = df.withColumn(c, _sql_try_cast(c, "int"))
        elif kind == "int":
            inteiros.append(c)
            df = df.withColumn(c, F.regexp_replace(_to_numeric_string(c), r"\.0+$", ""))
            df = df.withColumn(c, _sql_try_cast(c, "int"))
        else:
            textos.append(c)
            if c not in META_COLS:
                df = df.withColumn(c, F.col(c).cast(StringType()))
    if "header_row" in df.columns:
        df = df.withColumn("header_row", _sql_try_cast("header_row", "int"))
    log("Tipos aplicados:")
    log("  decimal(18,2): " + str(decimais))
    log("  int:           " + str(inteiros))
    log("  flag->int:     " + str(flags))
    log("  string:        " + str(len(textos)) + " colunas")
    return df


def to_local_path(path):
    return path.replace("dbfs:", "", 1) if path.startswith("dbfs:") else path


def copy_to_local_disk(volume_path, file_name):
    local_copy = os.path.join(LOCAL_TMP, file_name)
    if os.path.exists(local_copy):
        os.remove(local_copy)
    with open(to_local_path(volume_path), "rb") as src, open(local_copy, "wb") as dst:

    return local_copy

def full_table_name(table):
    return catalog_name + "." + schema_name + "." + table


def quoted_table_name(table):
    return "`" + catalog_name + "`.`" + schema_name + "`.`" + table + "`"


def list_tables_in_schema():
    nomes = []
    for r in spark.sql("SHOW TABLES IN " + catalog_name + "." + schema_name).collect():
        d = r.asDict()
        nome = d.get("tableName") or d.get("table") or d.get("name")
        if nome:
            nomes.append(nome)
    return nomes


def table_exists(table):
    return table in set(list_tables_in_schema())


def wipe_table(table):
    fqn = full_table_name(table)
    qn = quoted_table_name(table)
    if not table_exists(table):
        log("Tabela nao existe (nada a limpar): " + fqn)
        return
    try:
        antes = spark.table(fqn).count()
    except Exception:
        antes = None
    log("A limpar " + fqn + ((" (linhas atuais=" + "{:,}".format(antes) + ")") if antes is not None else ""))
    for stmt in ("DROP TABLE IF EXISTS " + qn, "DROP TABLE IF EXISTS " + fqn):
        try:
            log("SQL: " + stmt)
            spark.sql(stmt)
            if not table_exists(table):
                break
        except Exception as exc:
            log("DROP falhou: " + str(exc).split("\n")[0][:220])
    if table_exists(table):
        for stmt in ("DELETE FROM " + qn, "TRUNCATE TABLE " + qn):
            try:
                log("SQL: " + stmt)
                spark.sql(stmt)
            except Exception as exc:
                log("Limpeza alternativa falhou: " + str(exc).split("\n")[0][:220])
    if table_exists(table):
        restantes = spark.table(fqn).count()
        log("Tabela ainda existe apos limpeza. Linhas restantes: " + "{:,}".format(restantes))
        if restantes > 0:
            raise Exception("Nao foi possivel limpar " + fqn + " (" + "{:,}".format(restantes) + " linhas).")
    else:
        log("Tabela apagada: " + fqn)

def detect_header_row(pdf_raw):
    scan = min(len(pdf_raw), HEADER_SCAN_ROWS)
    best_row, best_score = 0, -1
    for r in range(scan):
        filled = textish = 0
        for v in pdf_raw.iloc[r].tolist():
            if v is None or (isinstance(v, float) and pd.isna(v)):
                continue
            text = str(v).strip()
            if text == "" or text.lower() == "nan":
                continue
            filled += 1
            if isinstance(v, str):
                textish += 1
        score = filled * 2 + textish
        if score > best_score:
            best_score, best_row = score, r
    return best_row


def forward_fill_headers(headers):
    filled, last = [], ""
    for h in headers:
        if h:
            last = h
            filled.append(h)
        else:
            filled.append(last)
    return filled


def read_excel_fast(local_copy, source_file, preferred_sheet):
    sheet_name = choose_sheet(local_copy, preferred_sheet)
    log("Sheet escolhida: " + sheet_name)
    with timed("read_excel " + source_file):
        pdf_raw = pd.read_excel(
            local_copy, sheet_name=sheet_name, header=None, engine=EXCEL_ENGINE, dtype=object
        )
    if pdf_raw.empty:
        raise Exception("Ficheiro / sheet vazia: " + source_file)
    header_row = detect_header_row(pdf_raw)
    log(
        "Header row=" + str(header_row)
        + " colunas=" + str(pdf_raw.shape[1])
        + " linhas_brutas=" + "{:,}".format(len(pdf_raw))
    )
    raw_headers = []
    for v in pdf_raw.iloc[header_row].tolist():
        raw_headers.append("" if v is None or (isinstance(v, float) and pd.isna(v)) else (decode_cell(v) or ""))
    raw_headers = forward_fill_headers(raw_headers)
    clean_headers = uniquify_names([sanitize_column_name(h) for h in raw_headers])
    pdf = pdf_raw.iloc[header_row + 1 :].copy()
    del pdf_raw
    pdf.columns = clean_headers
    pdf = pdf.dropna(how="all").reset_index(drop=True)
    with timed("decode colunas object"):
        for c in pdf.columns:
            if pdf[c].dtype == object:
                pdf[c] = decode_object_column(pdf[c])
    pdf = aplicar_regras_negocio(pdf, source_file)
    if CAST_ALL_TO_STRING:
        with timed("cast staging para string"):
            pdf = pdf.astype("string")
            pdf = pdf.where(pdf.notna(), None)
    pdf.attrs["sheet_name"] = sheet_name
    pdf.attrs["header_row"] = header_row + 1
    return pdf


def parquet_path_for(target_table, source_file):
    safe = re.sub(r"[^A-Za-z0-9_.-]+", "_", source_file)
    if safe.lower().endswith(".xlsx"):
        safe = safe[:-5]
    elif safe.lower().endswith(".xls"):
        safe = safe[:-4]
    return os.path.join(to_local_path(staging_folder), target_table, safe + ".parquet")


def stage_one_file(source_file, preferred_sheet, target_table, available_files):
    if source_file not in available_files:
        raise Exception("Ficheiro nao encontrado: " + source_file)

    out_path = parquet_path_for(target_table, source_file)
    if SKIP_EXISTING_PARQUET and os.path.exists(out_path) and os.path.getsize(out_path) > 0:
        log("SKIP Excel: parquet ja existe -> " + out_path)
        try:
            import pyarrow.parquet as pq
            pf = pq.ParquetFile(out_path)
            n_rows = pf.metadata.num_rows
            cols = [c for c in pf.schema_arrow.names if c not in ("source_file", "source_sheet", "header_row")]
            log("  retoma parquet: " + "{:,}".format(n_rows) + " linhas")
            return out_path, cols, n_rows, "", None
        except Exception as exc:
            log("Parquet existente ilegivel, a reprocessar Excel: " + str(exc).split("\n")[0][:180])

    local_copy = None
    try:
        with timed("copia local " + source_file):
            local_copy = copy_to_local_disk(available_files[source_file], source_file)
        pdf = read_excel_fast(local_copy, source_file, preferred_sheet)
        n_rows = len(pdf)
        cols = list(pdf.columns)
        log("Linhas lidas: " + "{:,}".format(n_rows) + "  campos: " + str(len(cols)))
        if n_rows == 0:
            raise Exception("Ficheiro sem linhas de dados: " + source_file)
        sheet_name = pdf.attrs.get("sheet_name", "")
        header_row = pdf.attrs.get("header_row", None)
        pdf["source_file"] = source_file
        pdf["source_sheet"] = sheet_name
        pdf["header_row"] = int(header_row) if header_row is not None else None
        os.makedirs(os.path.dirname(out_path), exist_ok=True)
        if os.path.exists(out_path):
            os.remove(out_path)
        with timed("to_parquet " + source_file):
            pdf.to_parquet(out_path, engine="pyarrow", index=False, compression="snappy")
        log("Parquet staging: " + out_path)
        return out_path, cols, n_rows, sheet_name, header_row
    finally:
        if local_copy and os.path.exists(local_copy):
            try:
                os.remove(local_copy)
            except Exception:
                pass
        gc.collect()


if spark.sql("SHOW CATALOGS").filter("catalog = '" + catalog_name + "'").count() == 0:
    raise Exception("Catalogo nao encontrado: " + catalog_name)
if spark.sql("SHOW SCHEMAS IN " + catalog_name).filter("databaseName = '" + schema_name + "'").count() == 0:
    raise Exception("Schema nao encontrado: " + schema_name)
log("Catalogo e schema validados.")

available_files = {f.name: f.path for f in dbutils.fs.ls(source_folder)}
log("Ficheiros no volume: " + str(len(available_files)))

drop_target_tables()
reset_staging()

by_table = defaultdict(list)
for item in FILE_MAPPINGS:
    by_table[item["table"]].append(item)

processed_tables, failed_files, schema_report = [], [], []
t_job = time.time()

for target_table, items in by_table.items():
    log("")
    log("=" * 80)
    log("TABELA " + target_table)
    log("Ficheiros: " + str([i["file"] for i in items]))
    log("=" * 80)
    parquet_files, loaded_files, n_rows_total = [], [], 0
    for item in items:
        source_file = item["file"]
        log("")
        log("-" * 80)
        log("PROCESSAR " + source_file + " -> " + target_table)
        log("-" * 80)
        try:
            out_path, cols, n_rows, sheet_name, header_row = stage_one_file(
                source_file, item.get("preferred_sheet"), target_table, available_files
            )
            parquet_files.append(out_path)
            loaded_files.append(source_file)
            n_rows_total += n_rows
            schema_report.append({
                "file": source_file,
                "table": full_table_name(target_table),
                "n_cols": len(cols),
                "n_rows": n_rows,
                "columns": cols,
            })
        except Exception as e:
            log("ERRO: " + source_file)
            log(str(e))
            failed_files.append((source_file, str(e)))
    if not parquet_files:
        log("Nenhum ficheiro valido para " + target_table + ".")
        continue

    fqn = full_table_name(target_table)
    log("A gravar " + "{:,}".format(n_rows_total) + " linhas de " + str(len(loaded_files)) + " ficheiro(s) em " + fqn)
    stage_uri = staging_folder.rstrip("/") + "/" + target_table
    with timed("spark.read.parquet + tipos + replace table " + target_table):
        df_out = aplicar_tipos_spark(spark.read.option("mergeSchema", "true").parquet(stage_uri))
        df_out = df_out.withColumn("load_timestamp", F.current_timestamp())
        tmp_view = "_tmp_load_" + target_table
        df_out.createOrReplaceTempView(tmp_view)
        wipe_table(target_table)
        stmt = "CREATE OR REPLACE TABLE " + quoted_table_name(target_table) + " USING DELTA AS SELECT * FROM " + tmp_view
        log("SQL: " + stmt)
        try:
            spark.sql(stmt)
        except Exception as exc:
            log("CREATE OR REPLACE falhou, fallback saveAsTable: " + str(exc).split("\n")[0][:220])
            df_out.write.format("delta").mode("overwrite").option("overwriteSchema", "true").saveAsTable(fqn)
        gravadas = spark.table(fqn).count()
        log("Confirmacao overwrite " + fqn + ": " + "{:,}".format(gravadas) + " gravadas / " + "{:,}".format(n_rows_total) + " lidas")
        if gravadas > n_rows_total * 2:
            raise Exception("Possivel duplicacao em " + fqn)
    log("Carga concluida: " + fqn)
    processed_tables.append(fqn)

log("")
