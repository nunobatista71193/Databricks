def timed(label):
    class _T:
        def __enter__(self):
            self.t0 = time.time()
            return self

        def __exit__(self, *args):
            log("  [tempo] " + label + ": " + "{:.1f}s".format(time.time() - self.t0))

    return _T()

# DECODE XML DO EXCEL (_x0020_ = espaco)
# ============================================================

_XML_ESCAPE_RE = re.compile(r"_x([0-9A-Fa-f]{4,8})_")
_XML_ESCAPE_LOOSE_RE = re.compile(r"(?<![A-Za-z0-9])x([0-9A-Fa-f]{4})(?![A-Za-z0-9])")


def decode_excel_xml_escapes(value):
    if value is None or not isinstance(value, str):
        return value

    text = value

    def _to_char(match):
        try:
            return chr(int(match.group(1), 16))
        except (ValueError, OverflowError):
            return match.group(0)

    text = _XML_ESCAPE_RE.sub(_to_char, text)

    if "x00" in text:
        text = _XML_ESCAPE_LOOSE_RE.sub(_to_char, text)

    text = text.replace("\xa0", " ")
    text = text.replace("\u200b", "")
    return text


def decode_cell(value):
    if isinstance(value, str):
        text = decode_excel_xml_escapes(value).strip()
        if text == "":
            return None
        return text
    return value


def _needs_xml_decode(text):
    return ("_x" in text) or ("x00" in text) or ("\xa0" in text) or ("\u200b" in text)


def decode_cell_fast(value):
    if not isinstance(value, str):
        return value
    if not _needs_xml_decode(value):
        text = value.strip()
        return text if text else None
    return decode_cell(value)


def decode_object_column(series):
    if series.dtype != object:
        return series

    sample = series.dropna().head(200)
    has_str = False
    has_escape = False
    for v in sample.tolist():
        if isinstance(v, str):
            has_str = True
            if _needs_xml_decode(v):
                has_escape = True
                break

    if not has_str:
        return series

    if has_escape:
        return series.map(decode_cell_fast)

    def _strip_empty(v):
        if isinstance(v, str):
            t = v.strip()
            return t if t else None
        return v

    return series.map(_strip_empty)

def sanitize_column_name(name):
    nome = decode_excel_xml_escapes(str(name) if name is not None else "")
    nome = nome.strip()

    if nome.lower().startswith("unnamed") or nome.lower() in ("", "nan", "none", "nat"):
        return "coluna_sem_nome"

    nome = re.sub(r"[ ,;{}()\n\t\r=]", "_", nome)

    substituicoes = {
        "º": "o",
        "ª": "a",
        "%": "pct",
        "/": "_",
        "\\": "_",
        "#": "num",
        "€": "eur",
        "$": "usd",
        "-": "_",
        ".": "_",
        "?": "",
        "!": "",
        "*": "",
        "+": "mais",
        "&": "e",
        "'": "",
        '"': "",
    }

    for old, new in substituicoes.items():
        nome = nome.replace(old, new)

    nome = re.sub("_+", "_", nome)
    nome = nome.strip("_")

    if nome == "":
        nome = "coluna_sem_nome"

    if nome[0].isdigit():
        nome = "c_" + nome

    return nome


def uniquify_names(names):
    usados = {}
    resultado = []
    for nome in names:
        if nome not in usados:
            usados[nome] = 0
            resultado.append(nome)
        else:
            usados[nome] += 1
            resultado.append(nome + "_" + str(usados[nome]))
    return resultado


def is_ags_file(source_file):
    return "ags" in source_file.lower()


def is_ocs_file(source_file):
    return "ocs" in source_file.lower()


def rename_ags_cdpostal(columns):
    novas = []
    ja_usou_codigo = False

    for nome in columns:
        if nome.upper() == "CDPOSTAL" and not ja_usou_codigo:
            novas.append("CODIGO_POSTAL")
            ja_usou_codigo = True
        else:
            novas.append(nome)

    return uniquify_names(novas)


def normalizar_codigo_postal_series(series):
    text = series.map(
        lambda v: None
        if v is None or (isinstance(v, float) and pd.isna(v))
        else str(int(v))
        if isinstance(v, int)
        else str(v)
    )
    text = text.astype("string")
    text = text.str.strip()
    text = text.str.replace("\xa0", "", regex=False)
    text = text.str.replace(" ", "", regex=False)
    text = text.str.replace(r"\.0+$", "", regex=True)

    invalid = text.isna() | text.isin(["", "nan", "none", "nat", "NaN", "None", "NaT"])
    text = text.mask(invalid, pd.NA)

    out = pd.Series(pd.NA, index=series.index, dtype="string")
    m_fmt = text.str.match(r"^\d{4}-\d{3}$", na=False)
    m_7 = text.str.match(r"^\d{7}$", na=False)
    out = out.mask(m_fmt, text)
    out = out.mask(m_7, text.str.slice(0, 4) + "-" + text.str.slice(4, 7))
    return out.astype(object).where(out.notna(), None)


def aplicar_regras_negocio(pdf, source_file):
    if is_ags_file(source_file):
        pdf.columns = rename_ags_cdpostal(list(pdf.columns))
        log("Regra AGS aplicada: CDPOSTAL -> CODIGO_POSTAL")

    if is_ocs_file(source_file):
        alvo = None
        for c in pdf.columns:
            if c.upper() == "CODIGO_POSTAL":
                alvo = c
                break

        if alvo is None:
            log("AVISO OCS: coluna CODIGO_POSTAL nao encontrada. Sem transformacao de conteudo.")
        else:
            if alvo != "CODIGO_POSTAL":
                pdf = pdf.rename(columns={alvo: "CODIGO_POSTAL"})
                alvo = "CODIGO_POSTAL"
            pdf[alvo] = normalizar_codigo_postal_series(pdf[alvo])
            validos = pdf[alvo].notna().sum()
            log(
                "Regra OCS aplicada em CODIGO_POSTAL: "
                + "{:,}".format(int(validos))
                + " validos / "
                + "{:,}".format(len(pdf))
                + " linhas (formato XXXX-XXX)"
            )

    return pdf

def to_local_path(path):
    if path.startswith("dbfs:"):
        return path.replace("dbfs:", "", 1)
    return path


def copy_to_local_disk(volume_path, file_name):
    local_src = to_local_path(volume_path)
    local_copy = os.path.join(LOCAL_TMP, file_name)

    if os.path.exists(local_copy):
        os.remove(local_copy)

    with open(local_src, "rb") as src, open(local_copy, "wb") as dst:
        shutil.copyfileobj(src, dst, length=8 * 1024 * 1024)

    return local_copy

def list_sheets(local_copy):
    xl = pd.ExcelFile(local_copy, engine=EXCEL_ENGINE)
    try:
        return list(xl.sheet_names)
    finally:
        xl.close()


def choose_sheet(local_copy, preferred):
    sheets = list_sheets(local_copy)
    log("")
    log("SHEETS NO FICHEIRO")
    for s in sheets:
        log("  - " + s)

    if preferred and preferred in sheets:
        return preferred

    if not sheets:
        raise Exception("Workbook sem sheets.")

    return sheets[0]


def detect_header_row(pdf_raw):
    scan = min(len(pdf_raw), HEADER_SCAN_ROWS)
    best_row = 0
    best_score = -1

    for r in range(scan):
        row = pdf_raw.iloc[r]
        filled = 0
        textish = 0
        for v in row.tolist():
            if v is None or (isinstance(v, float) and pd.isna(v)):
                continue
            text = str(v).strip()
            if text == "" or text.lower() == "nan":
                continue
            filled += 1
            if isinstance(v, str):
                textish += 1
        score = filled * 2 + textish
        if score > best_score:
            best_score = score
            best_row = r

    return best_row


def forward_fill_headers(headers):
    filled = []
    last = ""
    for h in headers:
        if h:
            last = h
            filled.append(h)
        else:
            filled.append(last)
    return filled


def read_excel_fast(local_copy, source_file, preferred_sheet):
    sheet_name = choose_sheet(local_copy, preferred_sheet)
    log("Sheet escolhida: " + sheet_name)

    with timed("read_excel " + source_file):
        pdf_raw = pd.read_excel(
            local_copy,
            sheet_name=sheet_name,
            header=None,
            engine=EXCEL_ENGINE,
            dtype=object,
        )

    if pdf_raw.empty:
        raise Exception("Ficheiro / sheet vazia: " + source_file)

    header_row = detect_header_row(pdf_raw)
    log("Linha de cabecalho detetada (0-based pandas): " + str(header_row))
    log("Colunas fisicas no Excel: " + str(pdf_raw.shape[1]))
    log("Linhas brutas lidas: " + "{:,}".format(len(pdf_raw)))

    raw_headers = []
    for v in pdf_raw.iloc[header_row].tolist():
        if v is None or (isinstance(v, float) and pd.isna(v)):
            raw_headers.append("")
        else:
            raw_headers.append(decode_cell(v) or "")

    raw_headers = forward_fill_headers(raw_headers)

    log("Cabecalhos ORIGINAIS (x0020 descodificado):")
    for i, h in enumerate(raw_headers, start=1):
        log("  [" + str(i).zfill(3) + "] " + (h if h else "<vazio>"))

    clean_headers = uniquify_names([sanitize_column_name(h) for h in raw_headers])

    log("Cabecalhos FINAIS (Delta, antes regras):")
    for i, orig in enumerate(raw_headers, start=1):
        log(
            "  ["
            + str(i).zfill(3)
            + "] "
            + repr(orig)
            + "  ->  "
            + clean_headers[i - 1]
        )

    pdf = pdf_raw.iloc[header_row + 1 :].copy()
    del pdf_raw
    pdf.columns = clean_headers
    pdf = pdf.dropna(how="all")
    pdf = pdf.reset_index(drop=True)

    with timed("decode colunas object"):
        for c in pdf.columns:
            if pdf[c].dtype == object:
                pdf[c] = decode_object_column(pdf[c])

    pdf = aplicar_regras_negocio(pdf, source_file)
    pdf = pdf.where(pd.notnull(pdf), None)

    if CAST_ALL_TO_STRING:
        with timed("cast colunas para string"):
            for c in pdf.columns:
                pdf[c] = pdf[c].map(
                    lambda v: None
                    if v is None or (isinstance(v, float) and pd.isna(v))
                    else str(v)
                )

    pdf.attrs["sheet_name"] = sheet_name
    pdf.attrs["header_row"] = header_row + 1
    return pdf


def parquet_path_for(target_table, source_file):
    safe = re.sub(r"[^A-Za-z0-9_.-]+", "_", source_file)
    if safe.lower().endswith(".xlsx"):
        safe = safe[:-5]
    elif safe.lower().endswith(".xls"):
        safe = safe[:-4]
    return os.path.join(to_local_path(staging_folder), target_table, safe + ".parquet")


def stage_one_file(source_file, preferred_sheet, target_table, available_files):
    if source_file not in available_files:
        raise Exception(
            "Ficheiro nao encontrado: "
            + source_file
            + ". Nomes no volume: "
            + str(sorted(available_files.keys()))
        )

    local_copy = None
    try:
        file_path = available_files[source_file]
        with timed("copia local " + source_file):
            local_copy = copy_to_local_disk(file_path, source_file)
        log("Copia local: " + local_copy)

        pdf = read_excel_fast(local_copy, source_file, preferred_sheet)
        n_rows = len(pdf)
        cols = list(pdf.columns)
        log("Linhas lidas: " + "{:,}".format(n_rows))
        log("Campos criados: " + str(len(cols)))

        if n_rows == 0:
            raise Exception("Ficheiro sem linhas de dados: " + source_file)

        sheet_name = pdf.attrs.get("sheet_name", "")
        header_row = pdf.attrs.get("header_row", None)

        pdf["source_file"] = source_file
        pdf["source_sheet"] = sheet_name
        pdf["header_row"] = int(header_row) if header_row is not None else None

        out_path = parquet_path_for(target_table, source_file)
        os.makedirs(os.path.dirname(out_path), exist_ok=True)
        if os.path.exists(out_path):
            os.remove(out_path)

        with timed("to_parquet " + source_file):
            pdf.to_parquet(out_path, engine="pyarrow", index=False, compression="snappy")

        log("Parquet staging: " + out_path)
        return out_path, cols, n_rows, sheet_name, header_row
    finally:
        if local_copy and os.path.exists(local_copy):
            try:
                os.remove(local_copy)
            except Exception:
                pass
        gc.collect()
