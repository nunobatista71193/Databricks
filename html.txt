function normView(s){s=s.trim().toUpperCase().replace(/\s|\./g,'');if(/^OR[CÇ]/.test(s)||s==='BUDGET'||s==='BUD26')return'ORC';return s}

function parseCSV(txt){
  const lines=txt.split(/\r?\n/).map(l=>l.trim()).filter(Boolean);
  if(lines.length<2)throw new Error('Preciso de uma linha de cabeçalho e pelo menos uma linha de dados.');
  const dl=lines[0].includes('\t')?'\t':lines[0].includes(';')?';':',';
  const H=lines[0].split(dl).map(h=>h.trim().toLowerCase());
  const need=['view','brand','seg','cov','exposure','ep','claims','cc'];
  for(const n of need)if(!H.includes(n))throw new Error('Falta a coluna "'+n+'" no cabeçalho.');
  const ix=Object.fromEntries(need.map(n=>[n,H.indexOf(n)]));
  const num=(s,i,n)=>{let t=String(s).replace(/\s/g,'');if(dl!==','&&t.includes(','))t=t.replace(/\./g,'').replace(',','.');const v=parseFloat(t);if(!isFinite(v))throw new Error('Linha '+(i+1)+': valor inválido em "'+n+'".');return v};
  const out=lines.slice(1).map((l,i)=>{const c=l.split(dl);const r={view:normView(c[ix.view]||''),brand:(c[ix.brand]||'').trim(),seg:normSeg(c[ix.seg]||''),cov:normCov(c[ix.cov]||'')};
    if(!VIEWS.includes(r.view))throw new Error('Linha '+(i+2)+': view "'+c[ix.view]+'" não reconhecida (usa '+VIEWS.join(', ')+').');
    if(!SEGS.includes(r.seg))throw new Error('Linha '+(i+2)+': seg "'+c[ix.seg]+'" não reconhecida.');
    if(!COVS.includes(r.cov))throw new Error('Linha '+(i+2)+': cov "'+c[ix.cov]+'" não reconhecida (DP ou RC).');
    if(!r.brand)throw new Error('Linha '+(i+2)+': marca vazia.');
    r.exposure=num(c[ix.exposure],i+1,'exposure');r.ep=num(c[ix.ep],i+1,'ep');r.claims=num(c[ix.claims],i+1,'claims');r.cc=num(c[ix.cc],i+1,'cc');return r});
  return out;
